cmake_minimum_required(VERSION 3.16)
project(RoomLocateNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc features2d flann calib3d)
add_library(roomloc_core Native/LocalizerCore.cpp)
target_include_directories(roomloc_core PUBLIC Native ${OpenCV_INCLUDE_DIRS})
target_link_libraries(roomloc_core PUBLIC ${OpenCV_LIBS})
add_executable(roomloc_native tests/native_cli.cpp)
target_link_libraries(roomloc_native PRIVATE roomloc_core)
enable_testing()
add_test(NAME geometry COMMAND roomloc_native --geometry-selftest)
